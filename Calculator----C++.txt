#include <iostream>
using namespace std;

int main() {
    // Declare variables for the numbers and the operation
    double num1, num2;
    char operation;

    // Display the menu
    cout << "Simple Calculator" << endl;
    cout << "==================" << endl;
    cout << "Enter the first number: ";
    cin >> num1;

    cout << "Enter the second number: ";
    cin >> num2;

    // Display the list of operations
    cout << "Choose an operation:" << endl;
    cout << "1. Add (+)" << endl;
    cout << "2. Subtract (-)" << endl;
    cout << "3. Multiply (*)" << endl;
    cout << "4. Divide (/)" << endl;

    // Take the operation choice
    cout << "Enter operation (+, -, *, /): ";
    cin >> operation;

    // Perform the corresponding operation based on the user input
    switch (operation) {
        case '+':
            cout << "Result: " << num1 + num2 << endl;
            break;
        case '-':
            cout << "Result: " << num1 - num2 << endl;
            break;
        case '*':
            cout << "Result: " << num1 * num2 << endl;
            break;
        case '/':
            if (num2 != 0) {
                cout << "Result: " << num1 / num2 << endl;
            } else {
                cout << "Error! Division by zero is not allowed." << endl;
            }
            break;
        default:
            cout << "Invalid operation. Please choose one of +, -, *, or /." << endl;
            break;
    }

    return 0;
}
